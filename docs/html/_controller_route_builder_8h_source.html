59 static int HexCharToDecimal(char c) {
60 if (c >= '0' && c <= '9') return c - '0';
61 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
62 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
63 return 0;
64 }

66 static char DecodeHexByte(char h1, char h2) {
67 return static_cast<char>((HexCharToDecimal(h1) << 4) | HexCharToDecimal(h2));

70 static DotNetDupe::System::String UrlDecode(const DotNetDupe::System::String& input) {
71 const char* pStr = input.GetRawString();
72 DotNetDupe::System::String decoded("");
73 int i = 0;
74 while (pStr && pStr[i] != '\0') {
75 if (pStr[i] == '%' && pStr[i + 1] != '\0' && pStr[i + 2] != '\0') {
76 decoded += DecodeHexByte(pStr[i + 1], pStr[i + 2]);
77 i += 3;
78 } else if (pStr[i] == '+') {
79 decoded += ' ';
80 i++;
81 } else {
82 decoded += pStr[i++];
83 }
84 }
85 return decoded;
86 }

88 static DotNetDupe::System::String ExtractFromSubPathPattern(const DotNetDupe::System::SmartPointer<Http::HttpContext>& ctx, const DotNetDupe::System::String& subPath) {
89 int openBrace = subPath.IndexOf("{");
90 int closeBrace = subPath.IndexOf("}");
91 if (openBrace >= 0 && closeBrace > openBrace + 1) {
92 DotNetDupe::System::String paramName = subPath.Substring(openBrace + 1, closeBrace - openBrace - 1);
93 DotNetDupe::System::String val;
94 if (ctx->GetRequest()->GetRouteValues().TryGetValue(paramName, val) && !val.IsEmpty()) {
95 return UrlDecode(val);
96 }
97 if (ctx->GetRequest()->GetQuery().TryGetValue(paramName, val) && !val.IsEmpty()) {
98 return UrlDecode(val);
99 }
100 }
101 return DotNetDupe::System::String("");
102 }

104 static DotNetDupe::System::String ExtractRouteOrQueryParam(const DotNetDupe::System::SmartPointer<Http::HttpContext>& ctx, const DotNetDupe::System::String& subPath) {
105 DotNetDupe::System::String val;
106 if (ctx->GetRequest()->GetRouteValues().TryGetValue("id", val) && !val.IsEmpty()) {
107 return UrlDecode(val);
108 }
109 DotNetDupe::System::String pathVal = ExtractFromSubPathPattern(ctx, subPath);
110 if (!pathVal.IsEmpty()) return pathVal;
111 if (ctx->GetRequest()->GetQuery().TryGetValue("id", val) && !val.IsEmpty()) {
112 return UrlDecode(val);
113 }
114 auto queryKeys = ctx->GetRequest()->GetQuery().GetKeys();
115 if (queryKeys.GetLength() > 0 && ctx->GetRequest()->GetQuery().TryGetValue(queryKeys[0], val)) {
116 return UrlDecode(val);
117 }
118 return DotNetDupe::System::String("");
119 }

385 // Registers recorded routes onto WebApplication
386 void Register(const DotNetDupe::System::SmartPointer<Builder::WebApplication>& app) override {
387 for (int i = 0; i < m_routes.GetCount(); ++i) {
388 const auto& r = m_routes[i];
389 if (r.Method == "GET") {
390 app->MapGet(m_routePrefix + r.SubPath, [r, app](DotNetDupe::System::SmartPointer<Http::HttpContext> ctx) {
391 return r.Handler(app, ctx);
392 });
393 } else if (r.Method == "POST") {
394 app->MapPost(m_routePrefix + r.SubPath, [r, app](DotNetDupe::System::SmartPointer<Http::HttpContext> ctx) {
395 return r.Handler(app, ctx);
396 });
397 } else if (r.Method == "PUT") {
398 app->MapPut(m_routePrefix + r.SubPath, [r, app](DotNetDupe::System::SmartPointer<Http::HttpContext> ctx) {
399 return r.Handler(app, ctx);
400 });
401 } else if (r.Method == "DELETE") {
402 app->MapDelete(m_routePrefix + r.SubPath, [r, app](DotNetDupe::System::SmartPointer<Http::HttpContext> ctx) {
403 return r.Handler(app, ctx);
404 });
405 }
406 }

123 template <typename U>
124 struct JsonConverter<Collections::Generic::List<U>> {
125 static JsonElement Write(const Collections::Generic::List<U>& value) {
126 JsonElement arr(JsonValueKind::Array);
127 for (int i = 0; i < value.GetCount(); ++i) {
128 arr.AddArrayElement(JsonConverter<U>::Write(value[i]));
129 }
130 return arr;
131 }
132 static Collections::Generic::List<U> Read(const JsonElement& element) {
133 Collections::Generic::List<U> lst;
134 int iLen = element.GetArrayLength();
135 for (int i = 0; i < iLen; ++i) {
136 lst.Add(JsonConverter<U>::Read(element.GetArrayElement(i)));
137 }
138 return lst;
139 }
140 };

143 template <typename U>
144 struct JsonConverter<Collections::Generic::Dictionary<String, U>> {
145 static JsonElement Write(const Collections::Generic::Dictionary<String, U>& value) {
146 JsonElement obj(JsonValueKind::Object);
147 auto keys = value.GetKeys();
148 for (int i = 0; i < keys.GetLength(); ++i) {
149 obj.SetProperty(keys[i], JsonConverter<U>::Write(value[keys[i]]));
150 }
151 return obj;
152 }
153 static Collections::Generic::Dictionary<String, U> Read(const JsonElement& element) {
154 Collections::Generic::Dictionary<String, U> dict;
155 auto keys = element.GetPropertyNames();
156 for (int i = 0; i < keys.GetLength(); ++i) {
157 JsonElement prop;
158 if (element.TryGetProperty(keys[i], prop)) {
159 dict.Add(keys[i], JsonConverter<U>::Read(prop));
160 }
161 }
162 return dict;
163 }

276 template <typename U>
277 struct JsonConverter<Array<U>> {
278 static JsonElement Write(const Array<U>& value) {
279 JsonElement arr(JsonValueKind::Array);
280 for (int i = 0; i < value.GetLength(); ++i) {
281 arr.AddArrayElement(JsonConverter<U>::Write(value[i]));
282 }
283 return arr;
284 }
285 static Array<U> Read(const JsonElement& element) {
286 int iLen = element.GetArrayLength();
287 Array<U> arr(iLen);
288 for (int i = 0; i < iLen; ++i) {
289 arr[i] = JsonConverter<U>::Read(element.GetArrayElement(i));
290 }
291 return arr;
292 }

316 template <typename U>
317 struct JsonConverter<Collections::Generic::Dictionary<String, U>> {
318 static JsonElement Write(const Collections::Generic::Dictionary<String, U>& value) {
319 JsonElement obj(JsonValueKind::Object);
320 auto keys = value.GetKeys();
321 for (int i = 0; i < keys.GetLength(); ++i) {
322 obj.SetProperty(keys[i], JsonConverter<U>::Write(value[keys[i]]));
323 }
324 return obj;
325 }
326 static Collections::Generic::Dictionary<String, U> Read(const JsonElement& element) {
327 Collections::Generic::Dictionary<String, U> dict;
328 auto keys = element.GetPropertyNames();
329 for (int i = 0; i < keys.GetLength(); ++i) {
330 JsonElement prop;
331 if (element.TryGetProperty(keys[i], prop)) {
332 dict.Add(keys[i], JsonConverter<U>::Read(prop));
333 }
334 }
335 return dict;
336 }

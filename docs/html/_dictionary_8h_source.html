167 pNewEntries[i].next = pNewBuckets[bucket];
168 pNewBuckets[bucket] = i;
169 }
170 }
171 if (m_pBuckets) FreeCollectionBuffer(m_pBuckets);
172 if (m_pEntries) FreeCollectionBuffer(m_pEntries);
173 m_pBuckets = pNewBuckets;
174 m_pEntries = pNewEntries;
175 m_iCapacity = newCapacity;

178 int FindEntry(const TKey& key) const {

181 for (int i = m_pBuckets[hashCode % m_iCapacity]; i >= 0; i = m_pEntries[i].next) {
182 if (m_pEntries[i].hashCode == hashCode && m_pEntries[i].key == key) return i;

191 int targetBucket = hashCode % m_iCapacity;

193 for (int i = m_pBuckets[targetBucket]; i >= 0; i = m_pEntries[i].next) {
194 if (m_pEntries[i].hashCode == hashCode && m_pEntries[i].key == key) {
195 if (add) throw System::ArgumentException("An item with the same key has already been added.");
196 m_pEntries[i].value = value;

198 }
199 }

203 index = m_iFreeList;
204 m_iFreeList = m_pEntries[index].next;
205 m_iFreeCount--;

210 }
211 index = m_iCount;
212 m_iCount++;
213 }

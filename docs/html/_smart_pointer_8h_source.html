514 void SetupEnableSharedFromThis(U* pPtr) {
515 if constexpr (IsComplete<U>::value) {
516 if (pPtr != nullptr && m_pnRefCount != nullptr) {
517 if constexpr (std::is_base_of_v<EnableSharedFromThisBase, U>) {

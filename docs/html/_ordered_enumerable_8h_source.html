32 template <typename T, typename FKey>
33 struct KeyComparer : IElementComparer<T> {
34 FKey m_keySelector;
35 bool m_bDescending;
36
37 KeyComparer(const FKey& keySelector, bool bDesc)
38 : m_keySelector(keySelector), m_bDescending(bDesc) {}

40 int Compare(const T& a, const T& b) const override {
41 auto keyA = m_keySelector(a);
42 auto keyB = m_keySelector(b);
43 int res = (keyA < keyB) ? -1 : ((keyB < keyA) ? 1 : 0);
44 return m_bDescending ? -res : res;

54 template <typename T, typename FKey>
55 struct ChainedComparer : IElementComparer<T> {
56 IElementComparer<T>* m_pParent;
57 FKey m_keySelector;
58 bool m_bDescending;
59
60 ChainedComparer(IElementComparer<T>* pParent, const FKey& keySelector, bool bDesc)
61 : m_pParent(pParent ? pParent->Clone() : nullptr),
62 m_keySelector(keySelector),
63 m_bDescending(bDesc) {}
64
65 ~ChainedComparer() override {
66 delete m_pParent;
67 }
68
69 int Compare(const T& a, const T& b) const override {
70 if (m_pParent) {
71 int parentRes = m_pParent->Compare(a, b);
72 if (parentRes != 0) return parentRes;
73 }
74 auto keyA = m_keySelector(a);
75 auto keyB = m_keySelector(b);
76 int res = (keyA < keyB) ? -1 : ((keyB < keyA) ? 1 : 0);
77 return m_bDescending ? -res : res;
78 }
79
80 IElementComparer<T>* Clone() const override {
81 return new ChainedComparer<T, FKey>(m_pParent, m_keySelector, m_bDescending);
82 }

98 static int Partition(Collections::Generic::List<int>& indices, const Collections::Generic::List<T>& items, int low, int high, const IElementComparer<T>& comp) {
99 int mid = low + (high - low) / 2;
100 SwapInt(indices[mid], indices[high]);
101 int pivotIdx = indices[high];
102 int i = low - 1;
103 for (int j = low; j < high; ++j) {
104 int curIdx = indices[j];
105 int c = comp.Compare(items[curIdx], items[pivotIdx]);
106 if (c < 0 || (c == 0 && curIdx < pivotIdx)) {
107 i++;
108 SwapInt(indices[i], indices[j]);
109 }
110 }
111 SwapInt(indices[i + 1], indices[high]);
112 return i + 1;
113 }
114
116 static void QuickSort(Collections::Generic::List<int>& indices, const Collections::Generic::List<T>& items, int low, int high, const IElementComparer<T>& comp) {
117 while (low < high) {
118 int pi = Partition(indices, items, low, high, comp);
119 if (pi - low < high - pi) {
120 QuickSort(indices, items, low, pi - 1, comp);
121 low = pi + 1;
122 } else {
123 QuickSort(indices, items, pi + 1, high, comp);
124 high = pi - 1;
125 }
126 }
127 }

130 static Collections::Generic::List<T> Sort(const Collections::Generic::List<T>& items, const IElementComparer<T>& comp) {
131 int count = items.GetCount();
132 Collections::Generic::List<int> indices(count);
133 for (int i = 0; i < count; ++i) { indices.Add(i); }
134 if (count > 1) { QuickSort(indices, items, 0, count - 1, comp); }
135 Collections::Generic::List<T> result(count);
136 for (int i = 0; i < count; ++i) { result.Add(items[indices[i]]); }
137 return result;
138 }

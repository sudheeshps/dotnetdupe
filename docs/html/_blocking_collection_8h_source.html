133 bool TryTake(T& item, int iTimeoutMilliseconds = 0) {
134 m_csLock.Enter();
135
136 if (m_list.GetCount() == 0) {
137 if (m_bIsAddingCompleted) {
138 m_csLock.Leave();
139 return false;
140 }
141
142 if (iTimeoutMilliseconds == 0) {
143 m_csLock.Leave();
144 return false;
145 }
146
147 if (iTimeoutMilliseconds < 0) {
148 while (!m_bIsAddingCompleted && m_list.GetCount() == 0) {
149 m_cvTake.Wait(m_csLock);
150 }
151 } else {
152 if (!m_bIsAddingCompleted && m_list.GetCount() == 0) {
153 m_cvTake.Wait(m_csLock, iTimeoutMilliseconds);
154 }
155 }
156
157 if (m_list.GetCount() == 0) {
158 m_csLock.Leave();
159 return false;
160 }
161 }
162
163 item = m_list.GetFirst()->Value;
164 m_list.RemoveFirst();
165
166 if (m_iBoundedCapacity > 0) {
167 m_cvAdd.Pulse();
168 }
169
170 m_csLock.Leave();

46 // Algorithm: Binary Min-Heap Sift-Up (O(log N))
47 // Traverses upward from leaf to root, swapping the item with its parent
48 // at (index - 1) / 2 while child priority < parent priority.
49 void SiftUp(int index) {
50 while (index > 0) {
51 int parent = (index - 1) / 2;
52 if (m_lstItems[parent] > m_lstItems[index]) {
53 std::swap(m_lstItems[parent], m_lstItems[index]);
54 index = parent;
55 } else {
56 break;
57 }
58 }
59 }
60
61 // Algorithm: Binary Min-Heap Sift-Down (O(log N))
62 // Moves the top element down by iteratively comparing against the smaller
63 // of its left (2*i + 1) and right (2*i + 2) children until heap order is restored.
64 void SiftDown(int index) {
65 int count = m_lstItems.GetCount();
66 while (index * 2 + 1 < count) {
67 int smallest = index * 2 + 1;
68 int right = index * 2 + 2;
69 if (right < count && m_lstItems[smallest] > m_lstItems[right]) {
70 smallest = right;
71 }
72 if (m_lstItems[index] > m_lstItems[smallest]) {
73 std::swap(m_lstItems[index], m_lstItems[smallest]);
74 index = smallest;
75 } else {
76 break;
77 }
78 }
79 }

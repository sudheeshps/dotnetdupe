152 // Algorithm: Binary Search with Bitwise Complement
153 // Performs logarithmic O(log N) lookup. If the key is not found, returns bitwise
154 // complement (~low) of the index of the first element larger than the searched value.
155 int BinarySearch(const T& item) const {
156 int low = 0;
157 int high = m_iCount - 1;
158 while (low <= high) {
159 int mid = low + (high - low) / 2;
160 if (m_pData[mid] == item) return mid;
161 if (m_pData[mid] < item) {
162 low = mid + 1;
163 } else {
164 high = mid - 1;
165 }
166 }
167 return ~low;
168 }

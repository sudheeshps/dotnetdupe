157 String BitConverter::ToString(Array<byte>& value, int iStartIndex, int nLength) {
158 int nArrayLength = value.GetLength();
159 if (iStartIndex < 0 || (iStartIndex >= nArrayLength && nArrayLength > 0))
160 throw ArgumentOutOfRangeException("iStartIndex");
161 if (nLength < 0)
162 throw ArgumentOutOfRangeException("nLength");
163 if (iStartIndex + nLength > nArrayLength)
164 throw ArgumentException("iStartIndex + nLength > value.Length");
165
166 if (nLength == 0) return String("");
167
169 std::stringstream ss;
170 for (int iIndex = 0; iIndex < nLength; ++iIndex) {
171 if (iIndex > 0) ss << "-";
172 ss << std::uppercase << std::hex << std::setw(2) << std::setfill('0') << (int)value [iStartIndex + iIndex];
173 }

            void LoggerTextWriter::Write(const DotNetDupe::System::String& sValue) {
                /// Emit message line to underlying logger.
                if (!sValue.IsEmpty() && !m_pLogger.IsNull()) {
                    m_pLogger->Log(m_logLevel, sValue);
                }
            }

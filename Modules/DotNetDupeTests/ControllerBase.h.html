                        Response()->SetContentType("application/json");

                        if (spToken.IsNull() || !spToken->Verify(sSecretKey)) {

                        auto keys = payload.GetKeys();

                        }

                        Response()->SetContentType("application/json");
                        Response()->SetBody("{\"error\":\"Unauthorized - Token parsing failed\"}");
